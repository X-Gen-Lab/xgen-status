cmake_minimum_required(VERSION 3.24)
project(xgs_consumer LANGUAGES C)

set(XGS_CONSUMER_MODE status CACHE STRING "Requested public contract")
set(XGS_CONSUMER_SOURCE_DIR "" CACHE PATH "Explicit source checkout; empty uses installed package")
option(XGS_CONSUMER_CHECK_CXX "Also build a C++ consumer" OFF)

set(consume_strings 0)
set(components status)
if(XGS_CONSUMER_MODE STREQUAL "strings" OR
   XGS_CONSUMER_MODE MATCHES "^strings_preexisting_" OR
   XGS_CONSUMER_MODE STREQUAL "preexisting_incomplete")
    set(consume_strings 1)
    set(components strings)
endif()

if(XGS_CONSUMER_MODE MATCHES "^preexisting_(bad_version|bad_abi|missing_properties|bad_type)$")
    set(fault "${CMAKE_MATCH_1}")
    if(fault STREQUAL "bad_type")
        add_library(xgs::status STATIC IMPORTED GLOBAL)
    else()
        add_library(xgs::status INTERFACE IMPORTED GLOBAL)
    endif()
    if(NOT fault STREQUAL "missing_properties")
        set_target_properties(xgs::status PROPERTIES XGS_VERSION 0.1.0 XGS_ABI_VERSION 1)
    endif()
    if(fault STREQUAL "bad_version")
        set_target_properties(xgs::status PROPERTIES XGS_VERSION 0.2.0)
    elseif(fault STREQUAL "bad_abi")
        set_target_properties(xgs::status PROPERTIES XGS_ABI_VERSION 2)
    endif()
endif()

if(XGS_CONSUMER_MODE STREQUAL "strings_preexisting_bad_abi" OR
   XGS_CONSUMER_MODE STREQUAL "preexisting_incomplete")
    add_library(xgs::status INTERFACE IMPORTED GLOBAL)
    set_target_properties(xgs::status PROPERTIES XGS_VERSION 0.1.0 XGS_ABI_VERSION 1)
    if(XGS_CONSUMER_MODE STREQUAL "strings_preexisting_bad_abi")
        add_library(xgs::strings STATIC IMPORTED GLOBAL)
        set_target_properties(xgs::strings PROPERTIES XGS_VERSION 0.1.0 XGS_ABI_VERSION 2)
    endif()
endif()

if(XGS_CONSUMER_SOURCE_DIR)
    set(XGS_BUILD_TESTS OFF CACHE BOOL "Production consumer" FORCE)
    set(XGS_ENABLE_COVERAGE OFF CACHE BOOL "Production consumer" FORCE)
    set(XGS_BUILD_STRINGS "${consume_strings}" CACHE BOOL "Explicit string selection" FORCE)
    add_subdirectory("${XGS_CONSUMER_SOURCE_DIR}" dependency)
    if(XGS_CONSUMER_MODE MATCHES "preexisting_compatible$")
        add_subdirectory("${XGS_CONSUMER_SOURCE_DIR}" reused)
    endif()
elseif(XGS_CONSUMER_MODE STREQUAL "wrong_version")
    find_package(xgen_status 0.2.0 EXACT CONFIG REQUIRED COMPONENTS status)
elseif(XGS_CONSUMER_MODE STREQUAL "unknown")
    find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS unknown)
elseif(XGS_CONSUMER_MODE STREQUAL "optional")
    find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS status OPTIONAL_COMPONENTS unknown)
    if(xgen_status_unknown_FOUND)
        message(FATAL_ERROR "Unknown optional component was accepted")
    endif()
elseif(XGS_CONSUMER_MODE STREQUAL "optional_strings")
    find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS status OPTIONAL_COMPONENTS strings)
    if(xgen_status_strings_FOUND)
        message(FATAL_ERROR "Disabled optional strings must be unavailable")
    endif()
elseif(XGS_CONSUMER_MODE STREQUAL "default")
    find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED)
else()
    find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
    if(XGS_CONSUMER_MODE MATCHES "preexisting_compatible$")
        find_package(xgen_status 0.1.0 EXACT CONFIG REQUIRED COMPONENTS ${components})
    endif()
endif()

get_target_property(status_type xgs::status TYPE)
get_target_property(status_dependencies xgs::status INTERFACE_LINK_LIBRARIES)
get_target_property(status_sources xgs::status INTERFACE_SOURCES)
get_target_property(status_version xgs::status XGS_VERSION)
get_target_property(status_abi xgs::status XGS_ABI_VERSION)
if(NOT status_type STREQUAL "INTERFACE_LIBRARY" OR status_dependencies OR status_sources)
    message(FATAL_ERROR "status-only contract requires a type-only INTERFACE target")
endif()
if(NOT status_version VERSION_EQUAL "0.1.0" OR NOT status_abi STREQUAL "1")
    message(FATAL_ERROR "Version/ABI properties were not preserved")
endif()
set(consumer_target xgs::status)
if(consume_strings)
    if(NOT TARGET xgs::strings)
        message(FATAL_ERROR "Requested strings target is missing")
    endif()
    get_target_property(strings_type xgs::strings TYPE)
    get_target_property(strings_version xgs::strings XGS_VERSION)
    get_target_property(strings_abi xgs::strings XGS_ABI_VERSION)
    if(NOT strings_type STREQUAL "STATIC_LIBRARY" OR
       NOT strings_version VERSION_EQUAL "0.1.0" OR NOT strings_abi STREQUAL "1")
        message(FATAL_ERROR "String target type or version/ABI was not preserved")
    endif()
    set(consumer_target xgs::strings)
elseif(TARGET xgs::strings)
    message(FATAL_ERROR "status-only consumer must not import strings")
endif()
if(TARGET GTest::gtest OR TARGET GTest::gtest_main OR TARGET xgc::base)
    message(FATAL_ERROR "Production consumer must not import GoogleTest or core")
endif()

add_executable(consumer_c consumer.c)
set_target_properties(consumer_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_link_libraries(consumer_c PRIVATE ${consumer_target})
target_compile_definitions(consumer_c PRIVATE XGS_CONSUME_STRINGS=${consume_strings})
enable_testing()
add_test(NAME consumer_c_runs COMMAND consumer_c)

if(XGS_CONSUMER_CHECK_CXX)
    enable_language(CXX)
    add_executable(consumer_cpp consumer.cpp)
    set_target_properties(consumer_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    target_link_libraries(consumer_cpp PRIVATE ${consumer_target})
    target_compile_definitions(consumer_cpp PRIVATE XGS_CONSUME_STRINGS=${consume_strings})
    add_test(NAME consumer_cpp_runs COMMAND consumer_cpp)
endif()
