cmake_minimum_required(VERSION 3.24)
project(xgs_status_only_contract LANGUAGES C)

if(NOT XGS_SOURCE_DIR)
    message(FATAL_ERROR "Pass the explicit XGS_SOURCE_DIR")
endif()
set(XGS_BUILD_STRINGS OFF CACHE BOOL "Type-only consumer" FORCE)
add_subdirectory("${XGS_SOURCE_DIR}" dependency)
get_target_property(status_type xgs::status TYPE)
if(NOT status_type STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "status-only contract requires INTERFACE_LIBRARY; got ${status_type}")
endif()
if(TARGET xgs::strings)
    message(FATAL_ERROR "status-only contract must not import strings")
endif()
